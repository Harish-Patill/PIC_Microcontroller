trigger type: edge and level

unsigned char read_dkp(unsigned char trigger){
    static unsigned char once = 1;                                      // Retains state between function calls

    if (trigger == EDGE){                                               // Check for edge-triggered mode
        if((PORTC & 0x0F) != ALL_RELEASED && once == 1){                // Detect the key pressd
            once = 0;                                                   // Prevent repeated detection
            return PORTC & 0x0F;                                        // Return the key
        }
        else if((PORTC & 0x0F) == ALL_RELEASED){                        // Check if all keys are released
            once = 1;                                                   // Enable next key detection
        }
    }

    else if(trigger == LEVEL){                     // Level-triggered mode
        if((PORTC & 0x0F) != ALL_RELEASED){        // Is a key pressed?
            return PORTC & 0x0F;                   // Return it (every call!)
        }
    }
    
    return ALL_RELEASED;  // No key pressed
}

Timer0:

void init_timer0(void){
    
    // Configure everything first, enable the timer last
    T08BIT = 1;         // 8-bit mode
    T0CS = 0;           // internal clock source (Fosc/4)
    PSA = 1;            // 1 = prescaler bypassed, 0 = prescaler used

    TMR0 = 6;           // preload (8-bit mode uses TMR0L actually)
                        // TMR0L is what claude said would be better, else it needs to read and write for both bytes.

    TMR0IF = 0;         // clear any old overflow flag
    TMR0IE = 1;         // enable Timer0 overflow interrupt

    TMR0ON = 1;         // start timer
}

Timer1:

void init_timer1(void){
    
    // we can either directly assign value to T1CON or set values for each bits
    // T1CON = 0x80;       // 1000 0000 -> RD16=1, prescale 1:1, internal clock, timer OFF
    
    // so for the RD16 we have 3 methods to set the value where RD16 = 1 is not one of them
        // 1. use T1RD16
        // 2. T1CONbits.RD16 = 1;
        // 3. use the T1CON method
    
    
    //T1CONbits.RD16 = 1;
    T1RD16 = 1;         // 16-bit read/write mode
    
    TMR1CS = 0;         // internal clock (Fosc/4)
    T1CKPS1 = 0;        // prescale 1:1  (T1CKPS1:T1CKPS0 = 00)
    T1CKPS0 = 0;        // for 1:8 set both to 1

    TMR1 = 3036;        // preload (65536 - 62500)

    TMR1IF = 0;         // clear old overflow flag
    TMR1IE = 1;         // enable Timer1 interrupt

    TMR1ON = 1;         // start timer

}

void init_timer2(void){
    
    // 2 methods, either dierctly to the T2CON or to the inidvidual bits
    // T2CONbits.T2OUTPS = 4;      // postscale 1:5 (value = postscale - 1)
    // T2CONbits.T2CKPS = 1;       // prescale 1:4 (0 = 1:1, 1 = 1:4, 2 or 3 = 1:16)
    

    T2CON = 0x21;               // 0010 0001 -> postscale 1:5, prescale 1:4, timer OFF
    
    PR2 = 249;                  // period register
    TMR2 = 0;                   // preload=0
    
    TMR2IF = 0;                 // timer2 interrupt flag
    TMR2IE = 1;                 // timer2 interrupt enable

    T2CONbits.TMR2ON = 1;       // start the timer at last
}
